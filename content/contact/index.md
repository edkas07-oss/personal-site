<script>
(function() {
  var form = document.getElementById('ew-contact-form');
  var submitBtn = document.getElementById('ew-submit-btn');
  var btnText = document.getElementById('ew-btn-text');
  var alertBox = document.getElementById('ew-form-alert');
  var accessKeyInput = document.getElementById('ew-access-key');

  var COOLDOWN_SECONDS = 60;
  var STORAGE_KEY = 'ew_contact_last_submission';

  function showAlert(message, type) {
    alertBox.className = 'ew-form-alert ew-form-alert--' + type;
    alertBox.innerHTML = message;
    alertBox.style.display = 'block';
    alertBox.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
  }

  function hideAlert() {
    alertBox.style.display = 'none';
    alertBox.innerHTML = '';
  }

  function checkRateLimit() {
    var lastSent = localStorage.getItem(STORAGE_KEY);
    if (lastSent) {
      var diff = Math.floor((Date.now() - parseInt(lastSent, 10)) / 1000);
      if (diff < COOLDOWN_SECONDS) {
        var remaining = COOLDOWN_SECONDS - diff;
        showAlert('⏳ Harap tunggu <strong>' + remaining + ' detik</strong> lagi sebelum mengirim pesan berikutnya.', 'warning');
        return false;
      }
    }
    return true;
  }

  form.addEventListener('submit', function(e) {
    e.preventDefault();
    hideAlert();

    // SECURITY 2: Client Rate Limiting Check
    if (!checkRateLimit()) {
      return;
    }

    var name = document.getElementById('ew-name').value.trim();
    var email = document.getElementById('ew-email').value.trim();
    var topic = document.getElementById('ew-topic').value;
    var message = document.getElementById('ew-message').value.trim();

    if (!name) {
      showAlert('⚠️ Mohon masukkan Nama Lengkap Anda.', 'error');
      return;
    }

    var emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    if (!email || !emailPattern.test(email)) {
      showAlert('⚠️ Mohon masukkan format Alamat Email yang valid.', 'error');
      return;
    }

    if (!topic) {
      showAlert('⚠️ Mohon pilih Topik Diskusi.', 'error');
      return;
    }

    if (!message || message.length < 15) {
      showAlert('⚠️ Pesan minimal berisi 15 karakter.', 'error');
      return;
    }

    if (accessKeyInput.value === 'YOUR_ACCESS_KEY_HERE') {
      showAlert('⚠️ <strong>Access Key belum dikonfigurasi:</strong> Masukkan Web3Forms Access Key Anda di file <code>content/contact/index.md</code>.', 'warning');
      return;
    }

    var subjectInput = document.getElementById('ew-form-subject');
    if (subjectInput) {
      subjectInput.value = '[eddywiyatno.my.id] ' + topic + ' dari ' + name;
    }

    submitBtn.disabled = true;
    btnText.textContent = 'Mengirim Pesan...';

    var formData = new FormData(form);

    fetch(form.action, {
      method: 'POST',
      body: formData,
      headers: {
        'Accept': 'application/json'
      }
    })
    .then(function(response) {
      return response.json();
    })
    .then(function(data) {
      if (data.success) {
        localStorage.setItem(STORAGE_KEY, Date.now().toString());
        showAlert('✅ <strong>Pesan Berhasil Terkirim!</strong> Terima kasih telah menghubungi. Saya akan membalas pesan Anda ke <em>' + email + '</em> secepatnya.', 'success');
        form.reset();
      } else {
        showAlert('❌ Gagal mengirim pesan: ' + (data.message || 'Terjadi kesalahan pada server form. Silakan hubungi langsung via email.'), 'error');
      }
    })
    .catch(function(err) {
      showAlert('❌ Gagal terhubung ke server form. Silakan periksa koneksi internet Anda atau kirimkan email langsung ke <strong>edkas07@gmail.com</strong>.', 'error');
    })
    .finally(function() {
      submitBtn.disabled = false;
      btnText.textContent = 'Kirim Pesan →';
    });

  });

})();
</script>
